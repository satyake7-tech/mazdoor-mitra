let currentLang = localStorage.getItem('lang') || 'en';

function changeLanguage(lang) {
    currentLang = lang;
    localStorage.setItem('lang', lang);
    applyTranslations();
}

function applyTranslations() {
    const t = translations[currentLang] || translations.en;
    document.querySelectorAll('[data-i18n]').forEach(el => {
        const key = el.getAttribute('data-i18n');
        if (t[key]) el.textContent = t[key];
    });
    document.querySelectorAll('[data-i18n-placeholder]').forEach(el => {
        const key = el.getAttribute('data-i18n-placeholder');
        if (t[key]) el.placeholder = t[key];
    });
    document.getElementById('langSelect').value = currentLang;
}

function t(key) {
    return (translations[currentLang] && translations[currentLang][key]) || translations.en[key] || key;
}

    e.preventDefault();
    const phone = document.getElementById('loginPhone').value;
    let user;
    if (currentRole === 'labour') user = db.labour.find(l => l.phone === phone);
    else user = db.contractors.find(c => c.phone === phone);

// ============ ACCEPT/REJECT APPLICATION ============
function acceptApplication(appId) {
    const app = db.applications.find(a => a.id === appId);
    if (app) {
        app.status = 'Accepted';
        const job = db.jobs.find(j => j.id === app.jobId);
        
        // Create payment transaction
        const totalAmount = job.wage * job.days;
        db.transactions.push({
            id: Date.now(),
            type: 'Job Payment',
            labourId: app.labourId,
            contractorId: job.contractorId,
            jobId: job.id,
            amount: totalAmount,
            commission: totalAmount * 0.15,
            netAmount: totalAmount * 0.85,
            date: new Date().toLocaleDateString(),
            status: 'Pending'
        });
        
        saveDB();
        showToast('✅ Worker accepted! They will be notified.');
        loadContractorDashboard();
    }
}

function rejectApplication(appId) {
    const app = db.applications.find(a => a.id === appId);
    if (app) {
        app.status = 'Rejected';
        saveDB();
        showToast('Application rejected');
        loadContractorDashboard();
    }
}

function completeJob(jobId) {
    const job = db.jobs.find(j => j.id === jobId);
    if (job) {
        job.status = 'Completed';
        // Process payments
        db.transactions.forEach(tx => {
            if (tx.jobId === jobId && tx.status === 'Pending') {
                tx.status = 'Paid';
                const labour = db.labour.find(l => l.id === tx.labourId);
                if (labour) {
                    labour.wallet = (labour.wallet || 0) + tx.netAmount;
                }
                const contractor = db.contractors.find(c => c.id === tx.contractorId);
                if (contractor) {
                    contractor.totalSpent = (contractor.totalSpent || 0) + tx.amount;
                }
            }
        });
        saveDB();
        showToast('✅ Job completed! Payments processed.');
        loadContractorDashboard();
    }
}

    );
    
    const jobsList = document.getElementById('jobsList');
    if (matchingJobs.length === 0) {
        jobsList.innerHTML = '<div class="empty-state">📭 No jobs available. Check back soon!</div>';
        return;
    }
    
    jobsList.innerHTML = matchingJobs.map(job => {
        const applied = db.applications.find(a => a.jobId === job.id && a.labourId === currentUser.id);
        return `
            <div class="job-card">
                <h4>${job.title}</h4>
                <p>👷 ${job.skill}</p>
                <p>📍 ${job.location} (${job.pincode})</p>
                <p>📅 ${job.days} days | 👥 ${job.workers} workers needed</p>
                <p>📆 Start: ${job.startDate}</p>
                <p class="wage">💰 ₹${job.wage}/day (Total: ₹${job.wage * job.days})</p>
                <p>👔 ${job.contractorName}</p>
                ${job.description ? `<p>📝 ${job.description}</p>` : ''}
                ${applied ? 
                    `<button class="applied-btn">✓ ${applied.status}</button>` : 
                    `<button class="apply-btn" onclick="applyJob(${job.id})">Apply Now</button>`
                }
            </div>
        `;
    }).join('');
}

function loadContractorDashboard() {
    document.getElementById('contractorName').textContent = currentUser.name;
    document.getElementById('contractorCompany').textContent = currentUser.company || currentUser.location;
    
    const myJobs = db.jobs.filter(j => j.contractorId === currentUser.id);
    document.getElementById('myJobsCount').textContent = myJobs.length;
    
    const myJobIds = myJobs.map(j => j.id);
    const allApps = db.applications.filter(a => myJobIds.includes(a.jobId));
    document.getElementById('totalApps').textContent = allApps.length;
    document.getElementById('activeWorkers').textContent = allApps.filter(a => a.status === 'Accepted').length;
    document.getElementById('totalSpent').textContent = Math.round(currentUser.totalSpent || 0);
    
    const myJobsList = document.getElementById('myJobsList');
    if (myJobs.length === 0) {
        myJobsList.innerHTML = '<div class="empty-state">📋 No jobs posted yet</div>';
    } else {
        myJobsList.innerHTML = myJobs.map(job => {
            const apps = db.applications.filter(a => a.jobId === job.id);
            return `
                <div class="job-card">
                    <h4>${job.title} <span class="badge badge-${job.status === 'Active' ? 'active' : 'completed'}">${job.status}</span></h4>
                    <p>👷 ${job.skill} | 💰 ₹${job.wage}/day × ${job.days} days</p>
                    <p>📍 ${job.location}</p>
                    <p><strong>${apps.length} applications received</strong></p>
                    ${apps.map(a => `
                        <div style="background:#f8f9fa; padding:10px; margin:8px 0; border-radius:6px;">
                            <strong>${a.labourName}</strong> ⭐${a.labourRating}<br>
                            <small>📞 <a href="tel:${a.labourPhone}">${a.labourPhone}</a> | Status: <span class="badge badge-${a.status === 'Accepted' ? 'active' : a.status === 'Rejected' ? 'rejected' : 'pending'}">${a.status}</span></small>
                            ${a.status === 'Applied' ? `
                                <div style="margin-top:8px;">
                                    <button class="accept-btn" onclick="acceptApplication(${a.id})">✓ Accept</button>
                                    <button class="reject-btn" onclick="rejectApplication(${a.id})">✗ Reject</button>
                                </div>
                            ` : ''}
                        </div>
                    `).join('')}
                    ${job.status === 'Active' && apps.some(a => a.status === 'Accepted') ? 
                        `<button class="complete-btn" onclick="completeJob(${job.id})">✓ Mark Complete & Pay</button>` : ''}
                </div>
            `;
        }).join('');
    }
    
    filterLabour();
}

function filterLabour() {
    const search = document.getElementById('labourSearch')?.value.toLowerCase() || '';
    const skill = document.getElementById('skillFilter')?.value || '';
    let filtered = db.labour;
    if (search) filtered = filtered.filter(l => 
        l.name.toLowerCase().includes(search) || l.location.toLowerCase().includes(search)
    );
    if (skill) filtered = filtered.filter(l => l.skill === skill);
    
    const container = document.getElementById('availableLabour');
    if (!container) return;
    if (filtered.length === 0) {
        container.innerHTML = '<div class="empty-state">No workers found</div>';
    } else {
        container.innerHTML = filtered.map(l => `
            <div class="labour-card">
                <div class="labour-info">
                    <h4>${l.name} ${l.verified ? '✅' : ''}</h4>
                    <p>👷 ${l.skill} | ${l.experience} years exp</p>
                    <p>📍 ${l.location} | ⭐ ${l.rating} | 💰 ₹${l.wage}/day</p>
                </div>
                <a href="tel:${l.phone}" class="call-btn">📞 Call</a>
            </div>
        `).join('');
    }
}

function loadAdminData() {
    document.getElementById('totalLabour').textContent = db.labour.length;
    document.getElementById('totalContractors').textContent = db.contractors.length;
    document.getElementById('totalJobs').textContent = db.jobs.length;
    
    let totalRevenue = 0;
    db.transactions.forEach(tx => { if (tx.status === 'Paid') totalRevenue += tx.commission; });
    document.getElementById('totalRevenue').textContent = Math.round(totalRevenue);
    
    document.getElementById('allLabour').innerHTML = db.labour.length === 0 ? 
        '<div class="empty-state">No workers</div>' :
        db.labour.map(l => `
            <div class="labour-card">
                <div class="labour-info">
                    <h4>${l.name} (${l.phone}) ${l.verified ? '✅' : ''}</h4>
                    <p>${l.skill} | ${l.location} | ⭐ ${l.rating} | Wallet: ₹${Math.round(l.wallet || 0)}</p>
                </div>
            </div>
        `).join('');
    
    document.getElementById('allContractors').innerHTML = db.contractors.length === 0 ?
        '<div class="empty-state">No contractors</div>' :
        db.contractors.map(c => `
            <div class="labour-card">
                <div class="labour-info">
                    <h4>${c.name} (${c.phone})</h4>
                    <p>${c.company || 'No Company'} | ${c.location} | Spent: ₹${Math.round(c.totalSpent || 0)}</p>
                </div>
            </div>
        `).join('');
    
    document.getElementById('allJobs').innerHTML = db.jobs.length === 0 ?
        '<div class="empty-state">No jobs</div>' :
        db.jobs.map(j => `
            <div class="job-card">
                <h4>${j.title} <span class="badge badge-${j.status === 'Active' ? 'active' : 'completed'}">${j.status}</span></h4>
                <p>${j.skill} | ${j.workers} workers | ₹${j.wage}/day × ${j.days} days</p>
                <p>By: ${j.contractorName} | 📍 ${j.location}</p>
                <p class="wage">Total: ₹${j.wage * j.workers * j.days}</p>
                <p style="color:#28a745;"><strong>Commission (15%): ₹${Math.round(j.wage * j.workers * j.days * 0.15)}</strong></p>
            </div>
        `).join('');
    
    document.getElementById('allPayments').innerHTML = db.transactions.length === 0 ?
        '<div class="empty-state">No payments</div>' :
        db.transactions.map(tx => {
            const labour = db.labour.find(l => l.id === tx.labourId);
            return `
                <div class="job-card">
                    <h4>${tx.type} <span class="badge badge-${tx.status === 'Paid' ? 'active' : 'pending'}">${tx.status}</span></h4>
                    <p>Worker: ${labour ? labour.name : 'Unknown'}</p>
                    <p>Total: ₹${tx.amount} | Worker Got: ₹${Math.round(tx.netAmount)}</p>
                    <p style="color:#28a745;"><strong>Our Commission: ₹${Math.round(tx.commission)}</strong></p>
                    <small>${tx.date}</small>
                </div>
            `;
        }).join('');
}

    applyTranslations();
    if (currentUser && currentRole) {
        if (currentRole === 'labour') {
            showScreen('labourDashboard');
            loadLabourDashboard();

            loadContractorDashboard();

    }
